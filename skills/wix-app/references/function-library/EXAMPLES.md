## Complete Examples

### Basic Function Library

A string utilities library with sync, async, parameterized, and void functions.

**`src/extensions/site/function-libraries/string-utils/string-utils.ts`**:

```typescript
import type { Text, NumberType, WebUrl } from '@wix/editor-react-types';

export function capitalize(text: Text): Text {
  return text.charAt(0).toUpperCase() + text.slice(1).toLowerCase();
}

export function truncate(text: Text, maxLength: NumberType = 50): Text {
  if (text.length <= maxLength) {
    return text;
  }
  return text.slice(0, maxLength - 3) + '...';
}

export async function fetchText(url: WebUrl): Promise<Text> {
  const response = await fetch(url);
  return await response.text();
}

export function logMessage(message: Text): void {
  console.log(`[app] ${message}`);
}
```

### Complex Parameters

A product utilities library demonstrating complex object params, array params, textEnum, and optional parameters with defaults.

**`src/extensions/site/function-libraries/product-utils/product-utils.ts`**:

```typescript
import type { Text, NumberType } from '@wix/editor-react-types';

interface Product {
  name: Text;
  price: NumberType;
  quantity: NumberType;
}

export function calculateTotal(products: Product[]): NumberType {
  return products.reduce((total, product) => {
    return total + product.price * product.quantity;
  }, 0);
}

export function formatPrice(
  amount: NumberType,
  currency: 'USD' | 'EUR' | 'GBP' = 'USD',
): Text {
  const symbols: Record<string, string> = { USD: '$', EUR: '€', GBP: '£' };
  return `${symbols[currency]}${amount.toFixed(2)}`;
}

export function formatUser(user: { firstName: Text; lastName: Text; age: NumberType }): Text {
  return `${user.firstName} ${user.lastName} (${user.age})`;
}
```

### DOM Event Handlers

An interaction handlers library demonstrating DOM event parameters.

**`src/extensions/site/function-libraries/interaction-handlers/interaction-handlers.ts`**:

```typescript
import type { Text } from '@wix/editor-react-types';

export function handleClick(event: PointerEvent): void {
  console.log(`Clicked at (${event.clientX}, ${event.clientY})`);
}

export function handleKeyPress(event: KeyboardEvent): Text {
  return event.key;
}

export function onSearchChange(event: Event): Text {
  return (event.target as HTMLInputElement).value;
}

export function onToggleChange(event: Event): boolean {
  return (event.target as HTMLInputElement).checked;
}
```
